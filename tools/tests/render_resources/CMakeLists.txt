cmake_minimum_required(VERSION 3.16)
project(StackRenderResourceChecks LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
get_filename_component(STACK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(StackRenderResourceTests
    render_resource_tests.cpp
    "${STACK_ROOT}/src/Renderer/ViewportSectionStore.cpp")
target_include_directories(StackRenderResourceTests PRIVATE "${STACK_ROOT}/src")
target_compile_features(StackRenderResourceTests PRIVATE cxx_std_17)
target_link_libraries(StackRenderResourceTests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(StackRenderResourceTests PRIVATE /W4 /permissive-)
else()
    target_compile_options(StackRenderResourceTests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
foreach(case index ram disk lease capacity publication timing policy)
    add_test(NAME StackRenderer.Resources.${case} COMMAND StackRenderResourceTests ${case})
    set_tests_properties(StackRenderer.Resources.${case} PROPERTIES TIMEOUT 20)
endforeach()
